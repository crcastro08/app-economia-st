<script>
  let dMiembros = [], dConceptos = [];

  window.onload = function() {
    google.script.run.withSuccessHandler(datos => {
      llenarSelect('responsable', datos.responsables);
      
      dMiembros = datos.miembros;
      const datalist = document.getElementById('listaMiembros');
      dMiembros.forEach(m => {
        let option = document.createElement('option');
        option.value = m.nombre;
        datalist.appendChild(option);
      });

      dConceptos = datos.conceptos;
      llenarSelect('concepto', dConceptos.map(c => c.concepto));
      
      llenarSelect('td-cat-2', datos.categoriasLista);
    }).obtenerDatosIniciales();
    
    const hoy = new Date();
    const anio = hoy.getFullYear();
    const mes = String(hoy.getMonth() + 1).padStart(2, '0'); 
    const dia = String(hoy.getDate()).padStart(2, '0');      
    
    document.getElementById('fecha').value = `${anio}-${mes}-${dia}`;
  };

  function llenarSelect(id, lista) {
    const s = document.getElementById(id);
    if(!s) return;
    s.innerHTML = '<option value="">Seleccione...</option>';
    if(lista && lista.length > 0){
      lista.forEach(i => s.add(new Option(i, i)));
    }
  }

  function actualizarTodo() {
    const valorInput = document.getElementById('miembro').value;
    const m1 = dMiembros.find(x => x.nombre === valorInput);
    
    if (m1) {
      document.getElementById('correo').value = m1.correo;
      document.getElementById('td-cat-1').innerText = m1.categoria;
      document.getElementById('td-cuota-1').innerText = Math.round(m1.cuota || 0);
      document.getElementById('td-saldo-1').innerText = Math.round(m1.saldoPendiente || 0);
      document.getElementById('td-pend-1').innerText = Math.round(m1.cuotasPendientes || 0);
      
      if (m1.categoria === "Refrigerios") {
        document.getElementById('concepto').value = "Refrigerios";
      } else if (m1.categoria === "Primer Nivel") {
        document.getElementById('concepto').value = "Primer Nivel";
      } else {
        document.getElementById('concepto').value = "Membresía";
      }
      
      actualizarConcepto(); 
      actualizarLogicaCodigo();
    } else {
      document.getElementById('correo').value = "";
      document.getElementById('td-cat-1').innerText = "-";
      document.getElementById('td-cuota-1').innerText = "0";
      document.getElementById('td-saldo-1').innerText = "0";
      document.getElementById('td-pend-1').innerText = "0";
    }
    
    actualizarFila2();
  }

  function actualizarFila2() {
    const valorInput = document.getElementById('miembro').value;
    const catSeleccionada = document.getElementById('td-cat-2').value;
    
    const m2 = dMiembros.find(x => x.nombre === valorInput && x.categoria === catSeleccionada);
    
    if (m2) {
      document.getElementById('td-cuota-2').innerText = Math.round(m2.cuota || 0);
      document.getElementById('td-saldo-2').innerText = Math.round(m2.saldoPendiente || 0);
      document.getElementById('td-pend-2').innerText = Math.round(m2.cuotasPendientes || 0);
      document.getElementById('radioFila2').checked = true;
    } else {
      document.getElementById('td-cuota-2').innerText = "0";
      document.getElementById('td-saldo-2').innerText = "0";
      document.getElementById('td-pend-2').innerText = "0";
    }
    
    // --- NUEVO: Fuerzamos la actualización del código si la fila 2 cambia ---
    actualizarLogicaCodigo();
  }

  function actualizarConcepto() {
    const c = dConceptos.find(x => x.concepto === document.getElementById('concepto').value);
    document.getElementById('esCuota').value = c ? c.esCuota.toLowerCase() : "";
    actualizarLogicaCodigo();
  }

  // --- NUEVA FUNCIÓN: Identifica qué categoría está seleccionada según el radio button ---
  function obtenerCategoriaActiva() {
    if (document.getElementById('radioFila1').checked) {
      return document.getElementById('td-cat-1').innerText.trim();
    } else {
      return document.getElementById('td-cat-2').value;
    }
  }

  function actualizarLogicaCodigo() {
    const esCuota = document.getElementById('esCuota').value;
    const nombreMiembro = document.getElementById('miembro').value;
    
    // --- NUEVO: Extraemos la categoría usando la función creada arriba ---
    const catActiva = obtenerCategoriaActiva();
    
    // --- NUEVO: Ahora buscamos al miembro comparando NOMBRE y CATEGORÍA ---
    const miem = dMiembros.find(x => x.nombre === nombreMiembro && x.categoria === catActiva);
    const conc = dConceptos.find(x => x.concepto === document.getElementById('concepto').value);
    
    document.getElementById('codigo').value = (esCuota === "si") ? (miem ? miem.codigoRef : "") : (conc ? conc.codigoAux : "");
  }

  function enviar() {
    const btn = document.getElementById('btnGuardar');
    const responsableVal = document.getElementById('responsable').value;
    const miembroVal = document.getElementById('miembro').value.trim();
    const codigoVal = document.getElementById('codigo').value.trim();
    const conceptoVal = document.getElementById('concepto').value.trim();

    // 0. Validación de responsable seleccionado
    if (!responsableVal || responsableVal === "") {
      alert("Por favor, seleccione un responsable.");
      return;
    }

    // 1. Validación de miembro existente
    const existeMiembro = dMiembros.some(x => String(x.nombre).trim() === miembroVal);
    if (!existeMiembro) {
      alert("Por favor, seleccione un miembro válido de la lista.");
      return;
    }

    // --- 2. NUEVA VALIDACIÓN: Campo Código vacío con excepciones ---
    if (codigoVal === "" && conceptoVal !== "Especies" && conceptoVal !== "Coacna") {
      alert("El Campo Codigo esta vacio, es posible que el usuario no este registrado");
      return; // El "return" detiene el proceso y evita que se guarde el registro
    }
    // --------------------------------------------------------------

    btn.disabled = true; 
    btn.innerText = "Guardando...";

    const usaFila1 = document.getElementById('radioFila1').checked;
    const saldoAEnviar = usaFila1 ? document.getElementById('td-saldo-1').innerText : document.getElementById('td-saldo-2').innerText;
    const cuotaAEnviar = usaFila1 ? document.getElementById('td-cuota-1').innerText : document.getElementById('td-cuota-2').innerText;

    const objetoData = {
      fecha: document.getElementById('fecha').value,
      responsable: document.getElementById('responsable').value,
      miembro: miembroVal,
      correo: document.getElementById('correo').value,
      concepto: conceptoVal,
      esCuota: document.getElementById('esCuota').value,
      valor: document.getElementById('valor').value,
      formaDePago: document.getElementById('formaPago').value,
      saldo: saldoAEnviar,
      cuota: cuotaAEnviar,
      observaciones: document.getElementById('observaciones').value,
      codigo: codigoVal,
      enviarCorreo: document.getElementById('checkCorreo').checked
    };

    google.script.run
      .withSuccessHandler(m => { 
        alert(m); 
        btn.disabled = false; 
        btn.innerText = "Guardar Registro"; 
        
        document.getElementById('miembro').value = ""; 
        document.getElementById('valor').value = "";   
        
        actualizarTodo(); 
      })
      .withFailureHandler(e => { 
        alert("Error: " + e); 
        btn.disabled = false; 
        btn.innerText = "Guardar Registro"; 
      })
      .Guardar(objetoData);
  }
</script>
